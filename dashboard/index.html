T1

Distinct signer accounts per UTC day. An activity proxy: bots, relayers and several accounts per person all count.

Query & method
-- Transaction signers per UTC day: an activity proxy, not unique people.
-- Includes failed transactions and relayer accounts. One scan feeds three grains:
-- day (signers, transactions), month (accounts signing on 2+ days) and 12-month period
-- (transactions by the period's 10 largest signers). Partial November 2026 is excluded from month/period.
WITH per_signer_day AS (
  SELECT block_date, signer_account_id, COUNT(*) AS transactions
  FROM `bigquery-public-data.crypto_near_mainnet_us.transactions`
  WHERE block_date >= DATE '2024-11-01' AND block_date < DATE '2026-11-02'
  GROUP BY block_date, signer_account_id
), complete_months AS (
  SELECT * FROM per_signer_day WHERE block_date < DATE '2026-11-01'
), per_signer_month AS (
  SELECT DATE_TRUNC(block_date, MONTH) AS month, signer_account_id, COUNT(*) AS active_days
  FROM complete_months GROUP BY month, signer_account_id
), per_signer_period AS (
  SELECT IF(block_date < DATE '2025-11-01', DATE '2024-11-01', DATE '2025-11-01') AS period_start,
    signer_account_id, SUM(transactions) AS transactions
  FROM complete_months GROUP BY period_start, signer_account_id
), periods AS (
  SELECT period_start, SUM(transactions) AS transactions, COUNT(*) AS signers,
    ARRAY_AGG(transactions ORDER BY transactions DESC LIMIT 10) AS top10
  FROM per_signer_period GROUP BY period_start
)
SELECT 'day' AS grain, block_date AS period_start, SUM(transactions) AS transactions,
  COUNT(*) AS signers, CAST(NULL AS INT64) AS accounts_2plus_days, CAST(NULL AS INT64) AS top10_transactions
FROM per_signer_day GROUP BY block_date
UNION ALL
SELECT 'month', month, NULL, COUNT(*), COUNTIF(active_days >= 2), NULL
FROM per_signer_month GROUP BY month
UNION ALL
SELECT 'period', period_start, transactions, signers, NULL,
  (SELECT SUM(t) FROM UNNEST(top10) AS t)
FROM periods
ORDER BY grain, period_start;

Accounts signing on 2 or more days in a month

T1

Signer accounts active on at least two distinct UTC days in each complete calendar month.

Query & method
-- Transaction signers per UTC day: an activity proxy, not unique people.
-- Includes failed transactions and relayer accounts. One scan feeds three grains:
-- day (signers, transactions), month (accounts signing on 2+ days) and 12-month period
-- (transactions by the period's 10 largest signers). Partial November 2026 is excluded from month/period.
WITH per_signer_day AS (
  SELECT block_date, signer_account_id, COUNT(*) AS transactions
  FROM `bigquery-public-data.crypto_near_mainnet_us.transactions`
  WHERE block_date >= DATE '2024-11-01' AND block_date < DATE '2026-11-02'
  GROUP BY block_date, signer_account_id
), complete_months AS (
  SELECT * FROM per_signer_day WHERE block_date < DATE '2026-11-01'
), per_signer_month AS (
  SELECT DATE_TRUNC(block_date, MONTH) AS month, signer_account_id, COUNT(*) AS active_days
  FROM complete_months GROUP BY month, signer_account_id
), per_signer_period AS (
  SELECT IF(block_date < DATE '2025-11-01', DATE '2024-11-01', DATE '2025-11-01') AS period_start,
    signer_account_id, SUM(transactions) AS transactions
  FROM complete_months GROUP BY period_start, signer_account_id
), periods AS (
  SELECT period_start, SUM(transactions) AS transactions, COUNT(*) AS signers,
    ARRAY_AGG(transactions ORDER BY transactions DESC LIMIT 10) AS top10
  FROM per_signer_period GROUP BY period_start
)
SELECT 'day' AS grain, block_date AS period_start, SUM(transactions) AS transactions,
  COUNT(*) AS signers, CAST(NULL AS INT64) AS accounts_2plus_days, CAST(NULL AS INT64) AS top10_transactions
FROM per_signer_day GROUP BY block_date
UNION ALL
SELECT 'month', month, NULL, COUNT(*), COUNTIF(active_days >= 2), NULL
FROM per_signer_month GROUP BY month
UNION ALL
SELECT 'period', period_start, transactions, signers, NULL,
  (SELECT SUM(t) FROM UNNEST(top10) AS t)
FROM periods
ORDER BY grain, period_start;

Weekly mean of daily signers

T2

Usage series for T2: the mean of the seven daily signer counts in each week.

Query & method
-- Transaction signers per UTC day: an activity proxy, not unique people.
-- Includes failed transactions and relayer accounts. One scan feeds three grains:
-- day (signers, transactions), month (accounts signing on 2+ days) and 12-month period
-- (transactions by the period's 10 largest signers). Partial November 2026 is excluded from month/period.
WITH per_signer_day AS (
  SELECT block_date, signer_account_id, COUNT(*) AS transactions
  FROM `bigquery-public-data.crypto_near_mainnet_us.transactions`
  WHERE block_date >= DATE '2024-11-01' AND block_date < DATE '2026-11-02'
  GROUP BY block_date, signer_account_id
), complete_months AS (
  SELECT * FROM per_signer_day WHERE block_date < DATE '2026-11-01'
), per_signer_month AS (
  SELECT DATE_TRUNC(block_date, MONTH) AS month, signer_account_id, COUNT(*) AS active_days
  FROM complete_months GROUP BY month, signer_account_id
), per_signer_period AS (
  SELECT IF(block_date < DATE '2025-11-01', DATE '2024-11-01', DATE '2025-11-01') AS period_start,
    signer_account_id, SUM(transactions) AS transactions
  FROM complete_months GROUP BY period_start, signer_account_id
), periods AS (
  SELECT period_start, SUM(transactions) AS transactions, COUNT(*) AS signers,
    ARRAY_AGG(transactions ORDER BY transactions DESC LIMIT 10) AS top10
  FROM per_signer_period GROUP BY period_start
)
SELECT 'day' AS grain, block_date AS period_start, SUM(transactions) AS transactions,
  COUNT(*) AS signers, CAST(NULL AS INT64) AS accounts_2plus_days, CAST(NULL AS INT64) AS top10_transactions
FROM per_signer_day GROUP BY block_date
UNION ALL
SELECT 'month', month, NULL, COUNT(*), COUNTIF(active_days >= 2), NULL
FROM per_signer_month GROUP BY month
UNION ALL
SELECT 'period', period_start, transactions, signers, NULL,
  (SELECT SUM(t) FROM UNNEST(top10) AS t)
FROM periods
ORDER BY grain, period_start;

Weekly burn vs gross new supply

T3

Solid: burn, measured as block-to-block falls in total_supply. Dashed: gross new supply (change in total_supply + burn, which equals the rises, i.e. issuance). The burn fields (tokens_burnt + receipt_conversion_tokens_burnt) are only a comparison, because they include the 30% function-call reward paid to contracts.

Query & method
-- Weekly (Monday-start, UTC) block-to-block changes in total_supply (yoctoNEAR, FLOAT64).
-- nearcore: supply = previous + minted - burnt, minting only on an epoch's first block,
-- so falls measure burn and rises measure issuance net of that block's burn.
-- Blocks are linked by prev_block_hash; the window's first block has no parent and no change.
WITH b AS (
  SELECT block_date, block_height, block_hash, prev_block_hash, total_supply
  FROM `bigquery-public-data.crypto_near_mainnet_us.blocks`
  WHERE block_date >= DATE '2025-11-03' AND block_date < DATE '2026-11-02'
), changes AS (
  SELECT DATE_TRUNC(b.block_date, WEEK(MONDAY)) AS week_start, b.block_height,
    b.total_supply, b.total_supply - p.total_supply AS change
  FROM b LEFT JOIN b AS p ON b.prev_block_hash = p.block_hash
)
SELECT week_start, COUNT(*) AS blocks,
  COUNTIF(change IS NULL) AS unlinked_blocks,
  COUNTIF(change > 0) AS rising_blocks,
  FORMAT('%.17g', SUM(IF(change < 0, -change, 0))) AS supply_falls_yocto,
  FORMAT('%.17g', SUM(IF(change > 0, change, 0))) AS supply_rises_yocto,
  FORMAT('%.17g', ARRAY_AGG(total_supply ORDER BY block_height LIMIT 1)[OFFSET(0)]) AS opening_supply_yocto,
  FORMAT('%.17g', ARRAY_AGG(total_supply ORDER BY block_height DESC LIMIT 1)[OFFSET(0)]) AS closing_supply_yocto
FROM changes
GROUP BY week_start
ORDER BY week_start;

-- Weekly (Monday-start, UTC) recorded burn in yoctoNEAR (FLOAT64 sums):
-- receipt execution outcomes plus transaction-to-receipt conversion.
WITH outcomes AS (
  SELECT DATE_TRUNC(block_date, WEEK(MONDAY)) AS week_start, SUM(tokens_burnt) AS burnt, COUNTIF(tokens_burnt IS NULL) AS null_rows
  FROM `bigquery-public-data.crypto_near_mainnet_us.execution_outcomes`
  WHERE block_date >= DATE '2025-11-03' AND block_date < DATE '2026-11-02'
  GROUP BY week_start
), conversions AS (
  SELECT DATE_TRUNC(block_date, WEEK(MONDAY)) AS week_start, SUM(receipt_conversion_tokens_burnt) AS burnt, COUNTIF(receipt_conversion_tokens_burnt IS NULL) AS null_rows
  FROM `bigquery-public-data.crypto_near_mainnet_us.transactions`
  WHERE block_date >= DATE '2025-11-03' AND block_date < DATE '2026-11-02'
  GROUP BY week_start
)
SELECT week_start,
  FORMAT('%.17g', o.burnt) AS outcome_burn_yocto, o.null_rows AS outcome_null_rows,
  FORMAT('%.17g', c.burnt) AS conversion_burn_yocto, c.null_rows AS conversion_null_rows
FROM outcomes AS o FULL OUTER JOIN conversions AS c USING (week_start)
ORDER BY week_start;
